------------------------------



Average Shipping Days = 
AVERAGEX(
    FILTER(
        'gold fact_orders_cleaned_v2',
        'gold fact_orders_cleaned_v2'[date_quality_flag] = "valid"
    ),
    DATEDIFF(
        'gold fact_orders_cleaned_v2'[order_date].[Date],
        'gold fact_orders_cleaned_v2'[shipping_date].[Date],
        DAY
    )
)
